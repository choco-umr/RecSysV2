    <script>
              document.getElementById("registerForm").addEventListener("submit", async (event) => {
                  event.preventDefault();

                  const response = await fetch("http://localhost:8000/api/UserRoutes/register", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({
                      Firstname: document.getElementById("Firstname").value,
                      Lastname: document.getElementById("Lastname").value,
                      Username: document.getElementById("Username").value,
                      Password: document.getElementById("password").value,
                      Gender: "Male",
                      Age: Number(document.getElementById("Age").value),
                      PreferredCategories: [document.getElementById("categorySelect").value]
                    })
                  });

                  const data = await response.json();
                  const alertBox = document.getElementById("alertBox");
                  if (data.success) {
                    alertBox.classList.remove("alert-danger");
                    alertBox.classList.add("alert-success");
                    alertBox.textContent = "Success! Redirecting..."; // friendly message instead of raw JSON
                    alertBox.classList.remove("d-none");

                    const roleResponse = await fetch("http://localhost:8000/api/FrontendHelperRouter/", {
                      credentials: "include"
                    });
                    const roleData = await roleResponse.json();

                    if (roleData.role === "Researchers") {
                      window.location.href = "researcher/ResearcherHome.html";
                    } else if (roleData.role === "Admins") {
                      window.location.href = "admin/AdminHome.html";
                    } else {
                      window.location.href = "UserPages/UserHome.html";
                    }

                  } else {
                    alertBox.classList.remove("alert-success");
                    alertBox.classList.add("alert-danger");
                    alertBox.textContent = JSON.stringify(data);
                    alertBox.classList.remove("d-none");
                  }

                });
          </script>
